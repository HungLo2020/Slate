cmake_minimum_required(VERSION 3.16)
project(slate_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
# A minimum compatible API, never an EXACT runtime version requirement.
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick Widgets)
add_library(slate_qt STATIC bridge.cpp bridge.h resources.qrc)
target_link_libraries(slate_qt PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Widgets)
set_target_properties(slate_qt PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS slate_qt ARCHIVE DESTINATION lib)

option(SLATE_SMOKE_TEST "Build in-process offscreen GUI smoke driver" OFF)
if(SLATE_SMOKE_TEST)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    target_sources(slate_qt PRIVATE smoke.cpp)
    target_link_libraries(slate_qt PRIVATE Qt6::Test)
    target_compile_definitions(slate_qt PRIVATE SLATE_SMOKE_TEST)
endif()
